========================================================================
EXPERIMENT 1A - BIT STUFFING
========================================================================

#include <stdio.h>
#include <string.h>

#define MAX_INPUT 1000
#define MAX_OUTPUT 2100

int bitStuffing(const char input[], char output[], size_t output_size) {
    size_t i, j = 0;
    int consecutive_ones = 0;
    const char *flag = "01111110";

    if (strlen(flag) + strlen(input) + strlen(flag) + 1 > output_size)
        return 0;

    for (i = 0; flag[i] != '\0'; i++) output[j++] = flag[i];

    for (i = 0; input[i] != '\0'; i++) {
        if (input[i] != '0' && input[i] != '1') return 0;

        output[j++] = input[i];
        if (input[i] == '1') {
            consecutive_ones++;
            if (consecutive_ones == 5) {
                output[j++] = '0'; /* Insert a zero after five consecutive ones. */
                consecutive_ones = 0;
            }
        } else {
            consecutive_ones = 0;
        }
    }

    for (i = 0; flag[i] != '\0'; i++) output[j++] = flag[i];
    output[j] = '\0';
    return 1;
}

int main(void) {
    char input[MAX_INPUT], output[MAX_OUTPUT];

    printf("Enter binary data stream: ");
    if (scanf("%999s", input) != 1) {
        printf("Input error.\\n");
        return 1;
    }

    if (!bitStuffing(input, output, sizeof(output))) {
        printf("Invalid binary input or data too long.\\n");
        return 1;
    }

    printf("Stuffed frame: %s\\n", output);
    return 0;
}

========================================================================
EXPERIMENT 1B - CHARACTER STUFFING
========================================================================

#include <stdio.h>
#include <string.h>

#define MAX_INPUT 500
#define MAX_OUTPUT 1100

int charStuffing(const char data[], char flag, char esc,
                 char result[], size_t result_size) {
    size_t i, j = 0;
    size_t needed = 3; /* Start flag, end flag, and null terminator. */

    for (i = 0; data[i] != '\0'; i++)
        needed += (data[i] == flag || data[i] == esc) ? 2 : 1;

    if (needed > result_size) return 0;

    result[j++] = flag;
    for (i = 0; data[i] != '\0'; i++) {
        if (data[i] == flag || data[i] == esc)
            result[j++] = esc;
        result[j++] = data[i];
    }
    result[j++] = flag;
    result[j] = '\0';
    return 1;
}

int main(void) {
    char data[MAX_INPUT], result[MAX_OUTPUT];
    const char flag = 'F', esc = 'E';

    printf("Enter data string (no spaces): ");
    if (scanf("%499s", data) != 1) {
        printf("Input error.\\n");
        return 1;
    }

    if (!charStuffing(data, flag, esc, result, sizeof(result))) {
        printf("Input too long.\\n");
        return 1;
    }

    printf("Flag = %c, Escape = %c\\n", flag, esc);
    printf("Stuffed frame: %s\\n", result);
    return 0;
}

========================================================================
EXPERIMENT 2 - CRC-12, CRC-16, CRC-CCITT
========================================================================

#include <stdio.h>
#include <string.h>

#define MAX_DATA 1000
#define MAX_WORK 1100

int isBinary(const char *s) {
    size_t i;
    if (s[0] == '\0') return 0;
    for (i = 0; s[i] != '\0'; i++)
        if (s[i] != '0' && s[i] != '1') return 0;
    return 1;
}

int computeCRC(const char *data, const char *generator, char *checksum,
               size_t checksum_size) {
    char temp[MAX_WORK];
    size_t data_len = strlen(data), gen_len = strlen(generator), i, j;
    size_t total_len;

    if (gen_len < 2 || data_len + gen_len >= sizeof(temp) ||
        checksum_size < gen_len) return 0;

    strcpy(temp, data);
    for (i = 0; i < gen_len - 1; i++) temp[data_len + i] = '0';
    temp[data_len + gen_len - 1] = '\0';
    total_len = strlen(temp);

    for (i = 0; i <= total_len - gen_len; i++) {
        if (temp[i] == '1') {
            for (j = 0; j < gen_len; j++)
                temp[i + j] = (temp[i + j] == generator[j]) ? '0' : '1';
        }
    }

    memcpy(checksum, temp + total_len - (gen_len - 1), gen_len - 1);
    checksum[gen_len - 1] = '\0';
    return 1;
}

int main(void) {
    char data[MAX_DATA], checksum[32];
    int choice;
    /* Generator polynomials represented from highest-degree bit to constant term. */
    const char *crc12 = "1100000001111";       /* x^12 + x^11 + x^3 + x^2 + x + 1 */
    const char *crc16 = "11000000000000101";   /* x^16 + x^15 + x^2 + 1 */
    const char *ccitt = "10001000000100001";   /* CRC-16/CCITT: x^16 + x^12 + x^5 + 1 */

    printf("Enter data bits (binary, no spaces): ");
    if (scanf("%999s", data) != 1 || !isBinary(data)) {
        printf("Invalid binary input.\\n");
        return 1;
    }

    printf("\\nSelect generator polynomial:\\n");
    printf("1. CRC-12\\n2. CRC-16\\n3. CRC-CCITT\\nChoice: ");
    if (scanf("%d", &choice) != 1) {
        printf("Invalid choice.\\n");
        return 1;
    }

    switch (choice) {
        case 1:
            computeCRC(data, crc12, checksum, sizeof(checksum));
            printf("Generator: %s\\n", crc12);
            break;
        case 2:
            computeCRC(data, crc16, checksum, sizeof(checksum));
            printf("Generator: %s\\n", crc16);
            break;
        case 3:
            computeCRC(data, ccitt, checksum, sizeof(checksum));
            printf("Generator: %s\\n", ccitt);
            break;
        default:
            printf("Invalid choice. Select 1, 2, or 3.\\n");
            return 1;
    }

    printf("CRC checksum: %s\\n", checksum);
    printf("Codeword to transmit: %s%s\\n", data, checksum);
    return 0;
}

========================================================================
EXPERIMENT 3 - SLIDING WINDOW AND GO-BACK-N
========================================================================

#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main(void) {
    int window_size, total_frames, base = 0;
    int loss_percent;

    printf("Enter window size: ");
    if (scanf("%d", &window_size) != 1 || window_size <= 0) {
        printf("Window size must be positive.\\n");
        return 1;
    }

    printf("Enter total number of frames: ");
    if (scanf("%d", &total_frames) != 1 || total_frames <= 0) {
        printf("Number of frames must be positive.\\n");
        return 1;
    }

    printf("Enter simulated ACK-loss percentage (0-100): ");
    if (scanf("%d", &loss_percent) != 1 || loss_percent < 0 || loss_percent > 100) {
        printf("Percentage must be from 0 to 100.\\n");
        return 1;
    }

    srand((unsigned int)time(NULL));

    while (base < total_frames) {
        int end = base + window_size;
        int i, failed = -1;
        if (end > total_frames) end = total_frames;

        printf("\\n--- Sending window: frames %d to %d ---\\n", base, end - 1);
        for (i = base; i < end; i++)
            printf("Sending frame %d\\n", i);

        for (i = base; i < end; i++) {
            if ((rand() % 100) < loss_percent) {
                printf("ACK lost for frame %d.\\n", i);
                failed = i;
                break;
            }
            printf("ACK received for frame %d\\n", i);
        }

        if (failed >= 0) {
            printf("Go-Back-N: retransmit from frame %d.\\n", failed);
            base = failed;
        } else {
            base = end;
            printf("All frames in this window acknowledged; window slides forward.\\n");
        }
    }

    printf("\\nAll frames transmitted and acknowledged successfully.\\n");
    return 0;
}

========================================================================
EXPERIMENT 4 - DIJKSTRA SHORTEST PATH
========================================================================

#include <stdio.h>

#define MAX 10
#define INF 1000000000

void dijkstra(int graph[MAX][MAX], int n, int src) {
    int dist[MAX], visited[MAX] = {0}, i, count;

    for (i = 0; i < n; i++) dist[i] = INF;
    dist[src] = 0;

    for (count = 0; count < n; count++) {
        int u = -1, v;
        for (i = 0; i < n; i++)
            if (!visited[i] && (u == -1 || dist[i] < dist[u])) u = i;

        if (u == -1 || dist[u] == INF) break;
        visited[u] = 1;

        for (v = 0; v < n; v++) {
            if (graph[u][v] > 0 && !visited[v] &&
                dist[u] + graph[u][v] < dist[v]) {
                dist[v] = dist[u] + graph[u][v];
            }
        }
    }

    printf("\\nShortest distances from vertex %d (vertices numbered 0 to %d):\\n", src, n - 1);
    for (i = 0; i < n; i++) {
        if (dist[i] == INF) printf("Vertex %d: unreachable\\n", i);
        else printf("Vertex %d: %d\\n", i, dist[i]);
    }
}

int main(void) {
    int graph[MAX][MAX], n, src, i, j;

    printf("Enter number of vertices (1-%d): ", MAX);
    if (scanf("%d", &n) != 1 || n < 1 || n > MAX) {
        printf("Invalid number of vertices.\\n");
        return 1;
    }

    printf("Enter the %d x %d adjacency matrix (0 = no edge; weights must be non-negative):\\n", n, n);
    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) {
            if (scanf("%d", &graph[i][j]) != 1 || graph[i][j] < 0) {
                printf("Invalid edge weight.\\n");
                return 1;
            }
        }
    }

    printf("Enter source vertex (0-%d): ", n - 1);
    if (scanf("%d", &src) != 1 || src < 0 || src >= n) {
        printf("Invalid source vertex.\\n");
        return 1;
    }

    dijkstra(graph, n, src);
    return 0;
}

EXPERIMENT 5 - BROADCAST TREE USING BFS
========================================================================

#include <stdio.h>

#define MAX_HOSTS 20

int main(void) {
    int adj[MAX_HOSTS][MAX_HOSTS], visited[MAX_HOSTS] = {0};
    int queue[MAX_HOSTS], n, source, front = 0, rear = 0;
    int i, j, edge_count = 0;

    printf("Enter number of hosts (1-%d): ", MAX_HOSTS);
    if (scanf("%d", &n) != 1 || n < 1 || n > MAX_HOSTS) {
        printf("Invalid number of hosts.\\n");
        return 1;
    }

    printf("Enter adjacency matrix (0 = no connection, 1 = connection):\\n");
    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) {
            if (scanf("%d", &adj[i][j]) != 1 ||
                (adj[i][j] != 0 && adj[i][j] != 1)) {
                printf("Matrix entries must be 0 or 1.\\n");
                return 1;
            }
        }
    }

    printf("Enter source host (0-%d): ", n - 1);
    if (scanf("%d", &source) != 1 || source < 0 || source >= n) {
        printf("Invalid source host.\\n");
        return 1;
    }

    visited[source] = 1;
    queue[rear++] = source;
    printf("\\nBroadcast tree edges:\\n");

    while (front < rear) {
        int current = queue[front++];
        for (j = 0; j < n; j++) {
            if (adj[current][j] == 1 && !visited[j]) {
                printf("Host %d ---> Host %d\\n", current, j);
                visited[j] = 1;
                queue[rear++] = j;
                edge_count++;
            }
        }
    }

    if (edge_count == 0)
        printf("No other hosts were reachable from the source.\\n");
    if (edge_count < n - 1)
        printf("Warning: the network is disconnected; some hosts cannot be reached.\\n");

    return 0;
}

========================================================================
EXPERIMENT 6 - DISTANCE VECTOR ROUTING
========================================================================

#include <stdio.h>

#define MAX_NODES 20
#define INF 999

int main(void) {
    int cost[MAX_NODES][MAX_NODES];
    int dist[MAX_NODES][MAX_NODES];
    int next_hop[MAX_NODES][MAX_NODES];
    int n, i, j, k, changed;

    printf("Enter number of nodes (1-%d): ", MAX_NODES);
    if (scanf("%d", &n) != 1 || n < 1 || n > MAX_NODES) {
        printf("Invalid number of nodes.\\n");
        return 1;
    }

    printf("Enter cost matrix (use %d for no direct link):\\n", INF);
    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) {
            if (scanf("%d", &cost[i][j]) != 1 || cost[i][j] < 0) {
                printf("Costs must be non-negative integers.\\n");
                return 1;
            }
            if (i == j) cost[i][j] = 0;
            dist[i][j] = cost[i][j];
            if (i == j) next_hop[i][j] = i;
            else if (cost[i][j] < INF) next_hop[i][j] = j;
            else next_hop[i][j] = -1;
        }
    }

    /* Bellman-Ford distance-vector updates; only use reachable links. */
    do {
        changed = 0;
        for (i = 0; i < n; i++) {
            for (k = 0; k < n; k++) {
                if (i == k || cost[i][k] >= INF) continue;
                for (j = 0; j < n; j++) {
                    if (dist[k][j] >= INF) continue;
                    if (cost[i][k] + dist[k][j] < dist[i][j]) {
                        dist[i][j] = cost[i][k] + dist[k][j];
                        next_hop[i][j] = k;
                        changed = 1;
                    }
                }
            }
        }
    } while (changed);

    for (i = 0; i < n; i++) {
        printf("\\nRouting table for Node %d\\n", i + 1);
        printf("Destination\\tCost\\tNext hop\\n");
        for (j = 0; j < n; j++) {
            printf("%d\\t\\t", j + 1);
            if (dist[i][j] >= INF) {
                printf("INF\\t-\\n");
            } else if (i == j) {
                printf("%d\\t%d (self)\\n", dist[i][j], i + 1);
            } else {
                printf("%d\\t%d\\n", dist[i][j], next_hop[i][j] + 1);
            }
        }
    }

    return 0;
}
